        # اگر شماره دیگری است، شماره‌تان را در خط زیر جایگزین کنید:
        target_phone = "09376732409"  # یا شماره‌ای که با آن لاگین کردید
        
        target_user = db.query(User).filter(User.phone_number == target_phone).first()
        if target_user:
            target_user.role = UserRoleEnum.ADMIN
            db.commit()
            print(f"✅ نقش کاربر {target_user.full_name} ({target_phone}) به ADMIN تغییر یافت.")

            # اگر کاربری که الان لاگین هستید را می‌خواهید ادمین کنید:

# core/security.py
from datetime import datetime, timedelta
from passlib.context import CryptContext
import jwt
from fastapi import Depends, HTTPException, status
from fastapi.security import OAuth2PasswordBearer
from sqlalchemy.orm import Session
from core.database import get_db

# وارد کردن مدل کاربری برای بررسی نقش
from modules.auth.models import User, UserRoleEnum

SECRET_KEY = "Elevator_Super_Secret_Key_For_JWT_Token_2026"
ALGORITHM = "HS256"
ACCESS_TOKEN_EXPIRE_MINUTES = 60 * 24 * 7

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
oauth2_scheme = OAuth2PasswordBearer(tokenUrl="/api/v1/auth/login")

def verify_password(plain_password, hashed_password):
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password):
    return pwd_context.hash(password)

def create_access_token(data: dict):
    to_encode = data.copy()
    expire = datetime.utcnow() + timedelta(minutes=ACCESS_TOKEN_EXPIRE_MINUTES)
    to_encode.update({"exp": expire})
    return jwt.encode(to_encode, SECRET_KEY, algorithm=ALGORITHM)

# ۱. دسترسی پایه: هر کاربر لاگین شده (تکنسین، مدیر، ادمین)
def get_current_user(token: str = Depends(oauth2_scheme), db: Session = Depends(get_db)):
    credentials_exception = HTTPException(
        status_code=status.HTTP_401_UNAUTHORIZED,
        detail="توکن نامعتبر است یا منقضی شده، لطفا دوباره لاگین کنید",
        headers={"WWW-Authenticate": "Bearer"},
    )
    try:
        payload = jwt.decode(token, SECRET_KEY, algorithms=[ALGORITHM])
        user_id: str = payload.get("sub")
        if user_id is None:
            raise credentials_exception
    except jwt.PyJWTError:
        raise credentials_exception
        
    user = db.query(User).filter(User.id == user_id).first()
    if user is None or not user.is_active:
        raise credentials_exception
    return user

# ۲. دسترسی عملیاتی: فقط MANAGER و ADMIN
def get_manager_user(current_user: User = Depends(get_current_user)):
    if current_user.role not in [UserRoleEnum.ADMIN, UserRoleEnum.MANAGER]:
        raise HTTPException(status_code=403, detail="دسترسی غیرمجاز: فقط مدیران سیستم مجاز هستند.")
    return current_user

# ۳. دسترسی کامل: فقط ADMIN
def get_admin_user(current_user: User = Depends(get_current_user)):
    if current_user.role != UserRoleEnum.ADMIN:
        raise HTTPException(status_code=403, detail="دسترسی غیرمجاز: این عملیات حساس فقط برای مدیر کل (ADMIN) مجاز است.")
    return current_user

# modules/analytics/notifications_router.py
from fastapi import APIRouter, Depends
from sqlalchemy.orm import Session
from sqlalchemy import Column, String, Boolean, DateTime, ForeignKey, Enum as SQLEnum
# ایمپورت کردن نوع داده UUID برای هماهنگی با جدول users
from sqlalchemy.dialects.postgresql import UUID as PGUUID 
from core.database import get_db, Base
from core.security import get_current_user
from modules.auth.models import User
import uuid
import datetime
import enum

router = APIRouter()

# 1. تعریف مدل جدول اعلانات
class NotificationType(str, enum.Enum):
    ASSIGN = "assign"    
    CANCEL = "cancel"    
    SYSTEM = "system"    
    FINANCE = "finance"  

class Notification(Base):
    __tablename__ = "notifications"

    # هر دو فیلد id و user_id باید از نوع UUID باشند
    id = Column(PGUUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    user_id = Column(PGUUID(as_uuid=True), ForeignKey("users.id"), nullable=False) 
    title = Column(String, nullable=False)
    body = Column(String, nullable=False)
    type = Column(SQLEnum(NotificationType), default=NotificationType.SYSTEM)
    is_read = Column(Boolean, default=False)
    created_at = Column(DateTime, default=datetime.datetime.utcnow)

# 2. دریافت لیست پیام‌های کاربر لاگین شده
@router.get("/", summary="دریافت پیام‌ها و اعلانات من")
def get_my_notifications(
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):
    notifs = db.query(Notification).filter(
        Notification.user_id == current_user.id
    ).order_by(Notification.created_at.desc()).limit(30).all()
    
    result = []
    for n in notifs:
        diff = datetime.datetime.utcnow() - n.created_at
        hours = diff.total_seconds() // 3600
        time_str = f"{int(hours)} ساعت پیش" if hours > 0 else "لحظاتی پیش"

        result.append({
            "id": str(n.id),
            "title": n.title,
            "body": n.body,
            "type": n.type.value,
            "is_read": n.is_read,
            "time": time_str
        })

@router.get("/summary", summary="دریافت آمار کلی داشبورد شرکت و معوقات مالی")
def get_dashboard_summary(
    db: Session = Depends(get_db),
    manager: User = Depends(get_manager_user) 
):
    total_buildings = db.query(Building).filter(Building.is_deleted == False).count()
    active_contracts = db.query(Contract).filter(Contract.status == ContractStatusEnum.ACTIVE).count()

    total_market_debt = db.query(func.sum(Contract.total_debt)).scalar() or 0.0
    thirty_days_ago = datetime.utcnow() - timedelta(days=30)
    recent_income = db.query(func.sum(Payment.amount)).filter(Payment.created_at >= thirty_days_ago).scalar() or 0.0

    emergency_cases = db.query(ServiceReport).filter(ServiceReport.is_emergency == True).count()
    recent_services = db.query(ServiceReport).filter(ServiceReport.visit_time >= thirty_days_ago).count()

    overdue_contracts = []
    all_active = db.query(Contract).filter(Contract.status == ContractStatusEnum.ACTIVE).all()
    
    for contract in all_active:
        if contract.total_debt <= 0:
            continue

        last_payment = db.query(Payment).filter(Payment.contract_id == contract.id).order_by(Payment.payment_date.desc()).first()
        base_date = last_payment.payment_date if last_payment else contract.start_date
        
        days_passed = (date.today() - base_date).days
        is_overdue = False

        if contract.payment_frequency == PaymentFrequencyEnum.MONTHLY and days_passed > 30:
            is_overdue = True
        elif contract.payment_frequency == PaymentFrequencyEnum.QUARTERLY and days_passed > 90:
            is_overdue = True
            
        if is_overdue:
            overdue_contracts.append({
                "contract_id": str(contract.id),
                "employer_name": contract.employer_name,
                "building_name": contract.building.name if contract.building else "نامشخص",
                "debt_amount": contract.total_debt,
                "days_passed": days_passed,
                "frequency": "ماهانه" if contract.payment_frequency == PaymentFrequencyEnum.MONTHLY else "سه‌ماهه"
            })

    return {
        "business": {
            "total_buildings": total_buildings,
            "active_contracts": active_contracts
        },
        "finance": {
            "total_market_debt": total_market_debt,
            "income_last_30_days": recent_income,
            "overdue_contracts": overdue_contracts
        },
        "technical": {
            "emergency_rescues": emergency_cases,
            "services_last_30_days": recent_services
        }
    }

    return {
        "access_token": access_token, 
        "token_type": "bearer",
        "role": user.role.value,

    }

import uuid

from sqlalchemy.dialects.postgresql import UUID

from core.database import Base

class ContractStatusEnum(str, enum.Enum):
    ACTIVE = "ACTIVE"
    EXPIRED = "EXPIRED"
    CANCELED = "CANCELED"

# === اضافه شدن ساختار نحوه پرداخت ===
class PaymentFrequencyEnum(str, enum.Enum):
    MONTHLY = "MONTHLY"       # یک ماهه
    QUARTERLY = "QUARTERLY"   # سه ماهه
    FLEXIBLE = "FLEXIBLE"     # زمان نامشخص / منعطف

class Contract(Base):
    __tablename__ = "contracts"

    id = Column(UUID(as_uuid=True), primary_key=True, default=uuid.uuid4)
    building_id = Column(UUID(as_uuid=True), ForeignKey("buildings.id"), nullable=False)
    contract_number = Column(String, unique=True, index=True, nullable=True)
    
    employer_name = Column(String, nullable=False)

    employer_phone = Column(String, nullable=True)
    
    start_date = Column(Date, nullable=False)
    end_date = Column(Date, nullable=False)
    
    total_amount = Column(Float, default=0.0)
    monthly_fee = Column(Float, default=0.0)

    payment_frequency = Column(Enum(PaymentFrequencyEnum), default=PaymentFrequencyEnum.FLEXIBLE)

@router.post("/", response_model=schemas.PaymentResponse, status_code=status.HTTP_201_CREATED)
def register_payment(
    payment_data: schemas.PaymentCreate,
    db: Session = Depends(get_db),
    manager: User = Depends(get_manager_user)
):
    """ثبت پرداختی جدید و کسر خودکار از بدهی قرارداد (Atomic Transaction)"""
    contract = db.query(Contract).filter(Contract.id == payment_data.contract_id).first()
    if not contract:
        raise HTTPException(status_code=404, detail="قرارداد مورد نظر یافت نشد")

    new_payment = models.Payment(**payment_data.model_dump())
    db.add(new_payment)
    contract.total_debt -= payment_data.amount

    db.commit()
    db.refresh(new_payment)
    return new_payment

@router.get("/chart-data", summary="دیتای نمودار درآمد ۷ روز اخیر")
def get_chart_data(
    db: Session = Depends(get_db),
    manager: User = Depends(get_manager_user)
):
    today = date.today()
    chart_data = []
    max_amount = 0.0

    for i in range(6, -1, -1):
        target_date = today - timedelta(days=i)
        
        daily_payments = db.query(models.Payment).filter(
            models.Payment.payment_date >= datetime.combine(target_date, datetime.min.time()),
            models.Payment.payment_date <= datetime.combine(target_date, datetime.max.time())
        ).all()
        
        daily_sum = sum(p.amount for p in daily_payments)
        
        if daily_sum > max_amount:
            max_amount = daily_sum

        chart_data.append({
            "day": target_date.strftime("%m/%d"), 
            "amount": daily_sum
        })

    return {"chart_data": chart_data, "max_amount": max_amount}

    db: Session = Depends(get_db),
    manager: User = Depends(get_manager_user)
):

    writer = csv.writer(output, delimiter=",")
    
    output.write('\ufeff') # پشتیبانی از زبان فارسی در اکسل
    writer.writerow(["شناسه تراکنش", "نام کارفرما", "مبلغ (تومان)", "تاریخ", "روش پرداخت", "کد پیگیری"])

    for p in payments:
        contract = p.contract
        employer = contract.employer_name if contract else "نامشخص"
        method = "نقدی" if p.payment_method.value == "CASH" else ("چک" if p.payment_method.value == "CHECK" else "واریز")
        writer.writerow([str(p.id), employer, int(p.amount), p.payment_date, method, p.tracking_code or "-"])

    output.seek(0)
    return StreamingResponse(
        iter([output.getvalue()]),
        media_type="text/csv",
        headers={"Content-Disposition": "attachment; filename=financial_report.csv"}
    )

@router.post("/", response_model=schemas.ServiceReportResponse, status_code=status.HTTP_201_CREATED)
def create_service_report(
    report_data: schemas.ServiceReportCreate, 
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):
    """ثبت گزارش سرویس جدید (با پشتیبانی از چک‌لیست ۲۵ گانه و منطق شرایط اضطراری)"""

    if not elevator:

    contract = db.query(Contract).filter(
        Contract.building_id == elevator.building_id,
        Contract.status == ContractStatusEnum.ACTIVE

    # اعمال قوانین قرارداد فقط در صورتی که شرایط اضطراری نباشد
    if not report_data.is_emergency:
        if not contract:
            raise HTTPException(status_code=403, detail="هیچ قرارداد فعالی برای این ساختمان ثبت نشده است. اجازه ثبت گزارش ندارید.")

        if contract.end_date < date.today():
            raise HTTPException(status_code=403, detail=f"قرارداد این ساختمان در تاریخ {contract.end_date} منقضی شده است. لطفا جهت تمدید به مدیریت ارجاع دهید.")

    parts_data = report_data.parts_used
    report_dict = report_data.model_dump(exclude={"parts_used"}) 
    
    new_report = models.ServiceReport(technician_id=str(current_user.id), **report_dict)
    db.add(new_report)
    db.flush() 
    
    for part in parts_data:
        new_part = models.ServicePart(
            report_id=new_report.id,
            part_name=part.part_name,
            action_type=part.action_type,
            quantity=part.quantity,
            description=part.description
        )
        db.add(new_part)
    
    db.commit()
    db.refresh(new_report)
    return new_report

@router.post("/{report_id}/media", status_code=status.HTTP_201_CREATED)
async def upload_report_media(
    report_id: UUID,
    files: List[UploadFile] = File(...),
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):
    """آپلود همزمان چندین عکس یا فایل صوتی برای یک گزارش خاص"""
    report = db.query(models.ServiceReport).filter(models.ServiceReport.id == report_id).first()
    if not report:
        raise HTTPException(status_code=404, detail="گزارش مورد نظر یافت نشد")
        
    uploaded_records = []
    
    for file in files:
        if file.content_type.startswith("image/"):
            media_type = models.MediaTypeEnum.IMAGE
            folder = "images"
        elif file.content_type.startswith("audio/") or file.content_type in ["video/mp4", "video/webm"]:
            media_type = models.MediaTypeEnum.AUDIO
            folder = "audio"
        else:
            raise HTTPException(status_code=400, detail=f"فرمت فایل {file.filename} پشتیبانی نمی‌شود")
            
        file_path = await save_file(file, folder)
        
        new_media = models.ServiceMedia(
            report_id=report.id,
            file_path=file_path,
            media_type=media_type
        )
        db.add(new_media)
        uploaded_records.append(new_media)
        
    db.commit()
    return {"message": f"{len(uploaded_records)} فایل با موفقیت برای گزارش ضمیمه شد"}

# shared/storage.py
import os
import uuid
from fastapi import UploadFile
from pathlib import Path

# ساخت پوشه اصلی و زیرپوشه‌ها در کنار پروژه
UPLOAD_DIR = Path("uploads")
UPLOAD_DIR.mkdir(parents=True, exist_ok=True)
(UPLOAD_DIR / "images").mkdir(exist_ok=True)
(UPLOAD_DIR / "audio").mkdir(exist_ok=True)

async def save_file(file: UploadFile, folder: str) -> str:
    """ذخیره فایل در سرور و برگرداندن مسیر آن"""
    # استخراج پسوند فایل (مثل jpg یا mp3)
    ext = file.filename.split('.')[-1]
    # ساخت یک نام یکتا برای جلوگیری از تداخل اسم‌ها
    new_filename = f"{uuid.uuid4().hex}.{ext}"
    file_path = UPLOAD_DIR / folder / new_filename
    
    # ذخیره فایل در هارد سرور
    with open(file_path, "wb") as buffer:
        buffer.write(await file.read())
    
    # برگرداندن مسیر نسبی برای ذخیره در دیتابیس
    return str(file_path)
